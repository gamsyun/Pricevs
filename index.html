/* ---------------- 1. 通用 (强制统一基准单位，确保量纲一致) ---------------- */ function calcGeneral() { const p = parseFloat(document.getElementById('g-price').value); const q = parseFloat(document.getElementById('g-qty').value); const unitType = document.getElementById('g-unit').value; if (!p || !q) { document.getElementById('g-res').innerHTML = '等待输入...'; return; } let baseQty = q; let displayUnit = ''; // 强制换算为最小统一基准单位，确保比价时单位完全一致 if (unitType === 'L') { baseQty = q * 1000; // 1L = 1000ml displayUnit = 'ml'; } else if (unitType === 'ml') { baseQty = q; displayUnit = 'ml'; } else if (unitType === 'kg') { baseQty = q * 1000; // 1kg = 1000g displayUnit = 'g'; } else if (unitType === 'g') { baseQty = q; displayUnit = 'g'; } else { baseQty = q; displayUnit = '个/件'; } // 统一按基准单位计算单价 const unitPrice = p / baseQty; document.getElementById('g-res').innerHTML = `
【通用计算结果】
` + `
总价: ¥${fmt(p)}
` + `
基准单价: ¥${fmt(unitPrice, 4)} / ${displayUnit}
`; }
